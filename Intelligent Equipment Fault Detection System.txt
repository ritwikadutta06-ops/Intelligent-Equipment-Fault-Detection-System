
#include <LiquidCrystal.h>
#include <math.h>

// ==================================================
// INTELLIGENT EQUIPMENT FAULT DETECTION SYSTEM
// Signals and Systems (SSN) Project
// Arduino Uno | Tinkercad Simulation
// ==================================================

// LCD connections: RS, EN, D4, D5, D6, D7
LiquidCrystal lcd(7, 8, 5, 4, 3, 2);

// Output pins
const byte MOTOR_PIN = 9;
const byte GREEN_LED = 10;
const byte YELLOW_LED = 11;
const byte RED_LED = 12;
const byte BUZZER_PIN = 13;

// Analog inputs
// Fourth potentiometer controls LCD contrast
// directly and does not require an analog pin.
const byte AMP_PIN = A0;
const byte FREQ_PIN = A1;
const byte BEARING_PIN = A2;

// Signal processing configuration
const int N = 64;
const float FS = 256.0;
const float FC = 90.0;
const float PI_F = 3.14159265358979323846;

// Signal buffers
float realPart[N];
float imagPart[N];

// Butterworth filter coefficients
float b0, b1, b2, a1, a2;

// Extracted features
float rmsValue = 0;
float peakValue = 0;
float dominantFreq = 0;
float bearingEnergy = 0;

// ==================================================
// SECOND-ORDER BUTTERWORTH LOW-PASS FILTER
// ==================================================

void setupFilter() {
  float k = tan(PI_F * FC / FS);

  float norm = 1.0 /
    (1.0 + sqrt(2.0) * k + k * k);

  b0 = k * k * norm;
  b1 = 2.0 * b0;
  b2 = b0;

  a1 = 2.0 * (k * k - 1.0) * norm;

  a2 = (1.0 - sqrt(2.0) * k + k * k)
       * norm;
}

void butterworthFilter() {
  float x1 = 0, x2 = 0;
  float y1 = 0, y2 = 0;

  for (int i = 0; i < N; i++) {
    float x = realPart[i];

    float y = b0 * x + b1 * x1 + b2 * x2
              - a1 * y1 - a2 * y2;

    realPart[i] = y;

    x2 = x1;
    x1 = x;
    y2 = y1;
    y1 = y;
  }
}

// ==================================================
// SIMULATED MOTOR VIBRATION SIGNAL
// ==================================================

void generateSignal(float amplitude,
                    float motorFreq,
                    float bearingLevel) {

  for (int i = 0; i < N; i++) {
    float t = i / FS;

    // Fundamental motor vibration
    float fundamental =
      amplitude *
      sin(2.0 * PI_F * motorFreq * t);

    // Bearing fault vibration at 72 Hz
    float bearing =
      bearingLevel *
      sin(2.0 * PI_F * 72.0 * t);

    // Simulated high-frequency noise
    float noise =
      0.08 *
      sin(2.0 * PI_F * 110.0 * t);

    realPart[i] =
      fundamental + bearing + noise;

    imagPart[i] = 0;
  }
}

// ==================================================
// TIME-DOMAIN FEATURE EXTRACTION
// ==================================================

void calculateFeatures() {
  float sumSquares = 0;
  peakValue = 0;

  for (int i = 0; i < N; i++) {
    float value = realPart[i];

    sumSquares += value * value;

    if (fabs(value) > peakValue) {
      peakValue = fabs(value);
    }
  }

  rmsValue = sqrt(sumSquares / N);
}

// ==================================================
// 64-POINT FAST FOURIER TRANSFORM
// ==================================================

void performFFT() {
  int j = 0;

  // Bit-reversal ordering
  for (int i = 1; i < N; i++) {
    int bit = N >> 1;

    while (j & bit) {
      j ^= bit;
      bit >>= 1;
    }

    j ^= bit;

    if (i < j) {
      float temp = realPart[i];
      realPart[i] = realPart[j];
      realPart[j] = temp;

      temp = imagPart[i];
      imagPart[i] = imagPart[j];
      imagPart[j] = temp;
    }
  }

  // FFT butterfly operations
  for (int len = 2; len <= N; len <<= 1) {
    float angle = -2.0 * PI_F / len;

    float wlenR = cos(angle);
    float wlenI = sin(angle);

    for (int i = 0; i < N; i += len) {
      float wr = 1.0;
      float wi = 0.0;

      for (int k = 0; k < len / 2; k++) {
        int u = i + k;
        int v = u + len / 2;

        float tr =
          wr * realPart[v] -
          wi * imagPart[v];

        float ti =
          wr * imagPart[v] +
          wi * realPart[v];

        float ur = realPart[u];
        float ui = imagPart[u];

        realPart[u] = ur + tr;
        imagPart[u] = ui + ti;

        realPart[v] = ur - tr;
        imagPart[v] = ui - ti;

        float nextWr =
          wr * wlenR - wi * wlenI;

        wi = wr * wlenI + wi * wlenR;
        wr = nextWr;
      }
    }
  }
}

// ==================================================
// FREQUENCY-DOMAIN FEATURE EXTRACTION
// ==================================================

void analyzeSpectrum() {
  float maxMagnitude = 0;

  dominantFreq = 0;
  bearingEnergy = 0;

  for (int k = 1; k < N / 2; k++) {
    float magnitude =
      sqrt(realPart[k] * realPart[k] +
           imagPart[k] * imagPart[k]);

    float frequency = k * FS / N;

    if (magnitude > maxMagnitude) {
      maxMagnitude = magnitude;
      dominantFreq = frequency;
    }

    // Bearing fault frequency band: 64-80 Hz
    if (frequency >= 64.0 &&
        frequency <= 80.0) {

      bearingEnergy +=
        magnitude * magnitude;
    }
  }

  bearingEnergy =
    4.0 * bearingEnergy / (N * N);
}

// ==================================================
// LED AND BUZZER INDICATORS
// ==================================================

void setIndicators(bool green,
                   bool yellow,
                   bool red,
                   bool alarm) {

  digitalWrite(GREEN_LED, green);
  digitalWrite(YELLOW_LED, yellow);
  digitalWrite(RED_LED, red);

  if (alarm) {
    tone(BUZZER_PIN, 1000);
  } else {
    noTone(BUZZER_PIN);
  }
}

// ==================================================
// SETUP
// ==================================================

void setup() {

  // DC motor control through TIP120
  pinMode(MOTOR_PIN, OUTPUT);
  digitalWrite(MOTOR_PIN, HIGH);

  // Indicators
  pinMode(GREEN_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);

  // Serial monitor
  Serial.begin(9600);

  // LCD initialization
  lcd.begin(16, 2);

  // Configure digital filter
  setupFilter();

  // Startup screen
  lcd.setCursor(0, 0);
  lcd.print("INTELLIGENT");

  lcd.setCursor(0, 1);
  lcd.print("FAULT DETECTOR");

  delay(2000);
  lcd.clear();

  Serial.println("================================");
  Serial.println("INTELLIGENT EQUIPMENT MONITOR");
  Serial.println("SIGNALS AND SYSTEMS PROJECT");
  Serial.println("================================");
}

// ==================================================
// MAIN LOOP
// ==================================================

void loop() {

  // Read potentiometer values
  int ampRaw = analogRead(AMP_PIN);
  int freqRaw = analogRead(FREQ_PIN);
  int bearingRaw = analogRead(BEARING_PIN);

  // Convert readings into signal parameters
  float amplitude =
    0.3 + 3.0 * ampRaw / 1023.0;

  float motorFreq =
    12.0 + 16.0 * freqRaw / 1023.0;

  float bearingLevel =
    2.5 * bearingRaw / 1023.0;

  // Keep motor running
  digitalWrite(MOTOR_PIN, HIGH);

  // Generate simulated vibration signal
  generateSignal(amplitude,
                 motorFreq,
                 bearingLevel);

  // Apply low-pass filter
  butterworthFilter();

  // Extract RMS and peak values
  calculateFeatures();

  // Perform FFT analysis
  performFFT();
  analyzeSpectrum();

  // Fault classification thresholds
  bool bearingFault =
    bearingEnergy > 0.25;

  bool imbalance =
    rmsValue > 1.25;

  // Update LCD
  lcd.clear();

  if (bearingFault) {

    // Bearing fault: Red LED and buzzer
    setIndicators(false, false, true, true);

    lcd.setCursor(0, 0);
    lcd.print("BEARING FAULT");

    Serial.println("STATUS: BEARING FAULT");
  }

  else if (imbalance) {

    // Motor imbalance: Yellow LED
    setIndicators(false, true, false, false);

    lcd.setCursor(0, 0);
    lcd.print("MOTOR IMBALANCE");

    Serial.println("STATUS: MOTOR IMBALANCE");
  }

  else {

    // Normal condition: Green LED
    setIndicators(true, false, false, false);

    lcd.setCursor(0, 0);
    lcd.print("MOTOR NORMAL");

    Serial.println("STATUS: NORMAL");
  }

  // Display RMS and dominant frequency
  lcd.setCursor(0, 1);

  lcd.print("R:");
  lcd.print(rmsValue, 2);

  lcd.print(" F:");
  lcd.print(dominantFreq, 0);

  // Print signal analysis results
  Serial.println("------------------------");

  Serial.print("Motor Frequency: ");
  Serial.print(motorFreq, 2);
  Serial.println(" Hz");

  Serial.print("RMS Value: ");
  Serial.println(rmsValue, 3);

  Serial.print("Peak Amplitude: ");
  Serial.println(peakValue, 3);

  Serial.print("Dominant Frequency: ");
  Serial.print(dominantFreq, 1);
  Serial.println(" Hz");

  Serial.print("Bearing Band Energy: ");
  Serial.println(bearingEnergy, 3);

  Serial.println("------------------------");

  delay(500);
}
